{{- /* A row of photos side by side, or a swipeable carousel. One image per line: "file.webp | alt text". Paths are relative to the page's photoDir param. Rows give each photo a width proportional to its aspect ratio, so every photo in the row shares one height and nothing is cropped. Usage: {{< photos >}} a.webp | Alt for a b.webp | Alt for b {{< /photos >}} {{< photos carousel=true >}} ... {{< /photos >}} */ -}} {{- $dir := .Page.Params.photoDir | default "" -}} {{- $c := .Get "carousel" -}} {{- $carousel := or (eq $c true) (eq $c "true") -}} {{- $items := slice -}} {{- range split (trim .Inner "\n ") "\n" -}} {{- $line := trim . " " -}} {{- if $line -}} {{- $parts := split $line "|" -}} {{- $src := printf "%s%s" $dir (trim (index $parts 0) " ") -}} {{- $alt := "" -}} {{- if gt (len $parts) 1 }}{{ $alt = trim (index $parts 1) " " }}{{ end -}} {{- $cfg := imageConfig (path.Join "static" $src) -}} {{- $items = $items | append (dict "src" $src "alt" $alt "w" $cfg.Width "h" $cfg.Height "ratio" (div (float $cfg.Width) $cfg.Height)) -}} {{- end -}} {{- end -}} {{- if $carousel -}} {{- else if eq (len $items) 1 -}} {{- $i := index $items 0 -}} {{- else -}} {{- end -}}
